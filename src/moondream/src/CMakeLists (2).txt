cmake_minimum_required(VERSION 3.0.2)
project(vlm_latency_limo)

find_package(catkin REQUIRED COMPONENTS
  rospy
  std_msgs
  sensor_msgs
  geometry_msgs
  nav_msgs
  cv_bridge
)

catkin_package()

catkin_install_python(PROGRAMS
  scripts/camera_lane_perception.py
  scripts/lane_control.py
  scripts/vlm_limo_image_sampler.py
  DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION}
)
